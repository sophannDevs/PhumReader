def svg(m):
    name=m.group(1); size=m.group(2) or '16'
    inner=ICONS[name]
    inner=re.sub(r'<(\w+)([^>]*?)\s*/>', r'<\1\2></\1>', inner)
    return '<svg width="%s" height="%s" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true">%s</svg>'%(size,size,inner)
def icons(t): return re.sub(r'\[\[i:([a-z-]+)(?::(\d+))?\]\]', svg, t)

def kmdict():
    d={}
    import glob as _g
    for f in sorted(_g.glob(os.path.join(SRC,'i18n','*.tsv'))):
        for line in open(f,encoding='utf-8'):
            line=line.rstrip('\n')
            if '\t' in line:
                a,b=line.split('\t',1)
                if a and b: d[a]=b
    return json.dumps(d,ensure_ascii=False)
def page(name, title, preview_h, chrome=True):
    d=os.path.join(SRC,name)
    body=open(d+'/body.html').read(); css=open(d+'/page.css').read() if os.path.exists(d+'/page.css') else ''
    js=open(d+'/page.js').read()
    props=json.loads(open(d+'/props.json').read()) if os.path.exists(d+'/props.json') else {}
    if chrome:
        body=body.replace('<!--HEADER-->',rd('header.html')).replace('<!--FOOTER-->',rd('footer.html')).replace('<!--LAYERS-->',rd('layers.html')).replace('<!--MOBNAV-->',rd('mobnav.html')).replace('<!--CNAV-->',rd('cnav.html'))
    p={"device":{"editor":None,"default":"desktop"}}
    p.update(props); p["$preview"]={"width":1440,"height":preview_h}
    out=HEAD%(title, rd('shared.css')+'\n'+css)+icons(body)+'\n</x-dc>\n<script type="text/x-dc" data-dc-script data-props=\''+json.dumps(p,ensure_ascii=False).replace("'","&#39;")+'\'>\nclass Component extends DCLogic {\n'+rd('shared.js').rstrip().replace('/*KMDICT*/{}',kmdict())+'\n\n  // ===== page =====\n'+js.rstrip()+'\n}\n</script>\n</body>\n</html>\n'
    open(OUT+'V2'+name+'.dc.html','w').write(out)
